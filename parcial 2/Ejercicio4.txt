#include <iostream>
#include <random>
using namespace std;

int main (){

    minstd_rand generador(random_device{}());
    uniform_int_distribution<int> venta(1, 1000);

    int ventas[100];
    int total = 0;

    for (int i = 0; i < 100; i++) {
        ventas[i] = venta(generador);
        total += ventas[i];
    }

    cout << "total: " << total << endl;

    return 0;
}
